#Commands to apply Hessian Rosenbrock
RUN: Window3d < input_m0.H n1=1 > x_tmp_rosenbrock.H
RUN: Window3d < input_m0.H n1=1 f1=1> y_tmp_rosenbrock.H
RUN: Window3d < input.H n1=1 > dmx_tmp_rosenbrock.H
RUN: Window3d < input.H n1=1 f1=1> dmy_tmp_rosenbrock.H
RUN: Math file1=x_tmp_rosenbrock.H file2=y_tmp_rosenbrock.H file3=dmx_tmp_rosenbrock.H file4=dmy_tmp_rosenbrock.H exp="(2.0-400.0*file2+1200.0*file1*file1)*file3-400.0*file1*file4" >tmp_out_rosen_x.H
RUN: Math file1=x_tmp_rosenbrock.H file2=y_tmp_rosenbrock.H file3=dmx_tmp_rosenbrock.H file4=dmy_tmp_rosenbrock.H exp="-400.0*file1*file3+200.0*file4" >tmp_out_rosen_y.H
RUN: Cat3d tmp_out_rosen_x.H tmp_out_rosen_y.H axis=1 >output.H
RUN: Rm x_tmp_rosenbrock.H y_tmp_rosenbrock.H dmx_tmp_rosenbrock.H dmy_tmp_rosenbrock.H tmp_out_rosen_x.H tmp_out_rosen_y.H
